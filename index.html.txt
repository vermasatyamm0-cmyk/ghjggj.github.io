function cleanVideoInput(value){
  let s = String(value == null ? '' : value).trim();
  if(!s) return '';
  const iframeMatch = s.match(/<iframe[^>]+(?:src|data-src)=[\"']([^\"']+)[\"'][^>]*>/i);
  if(iframeMatch && iframeMatch[1]) s = iframeMatch[1].trim();
  s = s.replace(/^[a-z]{1,4}(?=https?:\/\/)/i, '');
  s = s.replace(/^(?:url|link)\s*[:=]\s*/i, '');
  s = s.replace(/^['\"`\s]+|['\"`\s]+$/g, '');
  return s.trim();
}
function youtubeId(url){
  const u = String(url||'').trim();
  if(!u) return '';
  if(/^[A-Za-z0-9_-]{11}$/.test(u)) return u;
  try{
    const parsed = new URL(u, location.href);
    const host = parsed.hostname.toLowerCase().replace(/^www\./,'').replace(/^m\./,'').replace(/^music\./,'');
    if(host === 'youtu.be') return (parsed.pathname.split('/').filter(Boolean)[0] || '').split(/[?#]/)[0];
    if(host === 'youtube.com' || host === 'youtube-nocookie.com'){
      const v = parsed.searchParams.get('v');
      if(v && /^[A-Za-z0-9_-]{11}$/.test(v)) return v;
      const parts = parsed.pathname.split('/').filter(Boolean);
      if(['embed','shorts','live','v','e'].includes(parts[0])) {
        const id = (parts[1]||'').split(/[?#]/)[0];
        if(/^[A-Za-z0-9_-]{11}$/.test(id)) return id;
      }
    }
  }catch(e){}
  const m = u.match(/(?:youtu\.be\/|youtube(?:-nocookie)?\.com\/(?:watch\?(?:.*&)?v=|embed\/|shorts\/|live\/|v\/|e\/))([A-Za-z0-9_-]{11})/i);
  return m ? m[1] : '';
}
function vimeoId(url){
  const u = String(url||'').trim();
  if(!u) return '';
  const m = u.match(/vimeo\.com\/(?:video\/)?(\d+)/i);
  return m ? m[1] : '';
}

function safeUrl(url){
  const u = cleanVideoInput(url);
  if(!u) return '';
  if(/^(javascript|data|vbscript):/i.test(u)) return '';
  if(!/^https?:\/\//i.test(u) && !u.startsWith('//')) return '';
  return u.startsWith('//') ? 'https:' + u : u;
}
function embeddedSourceUrl(url){
  try{
    const parsed = new URL(String(url||''), location.href);
    const cand = parsed.searchParams.get('source')||parsed.searchParams.get('src')||parsed.searchParams.get('url')||parsed.searchParams.get('video');
    if(!cand) return '';
    const dec = decodeURIComponent(cand);
    return /^https?:\/\//i.test(dec) ? dec : '';
  }catch(e){ return ''; }
}

}
function normalizeVideoUrl(input){
  let raw = '';
  if(input && typeof input === 'object'){
    raw = firstValue(input, ['url','link','videoUrl','videoURL','videoLink','videoUri','src','video','embed','embedUrl','iframeUrl','source','sourceUrl','ytUrl','ytLink','youtubeUrl','youtubeLink'], '');
  }else{ raw = input || ''; }
  raw = cleanVideoInput(raw);
  if(!raw) return {src:'', type:'none', youtubeId:'', vimeoId:'', embedUrl:'', thumbnail:''};
  if(/^[A-Za-z0-9_-]{11}$/.test(raw)){
    return {src:'https://www.youtube.com/watch?v='+raw, type:'youtube', youtubeId: raw, vimeoId:'', embedUrl:'', thumbnail:`https://i.ytimg.com/vi/${encodeURIComponent(raw)}/hqdefault.jpg`};
  }
  const src = safeUrl(raw);
  if(!src) return {src:'', type:'none', youtubeId:'', vimeoId:'', embedUrl:'', thumbnail:''};
  const yid = youtubeId(src);
  if(yid) return {src, type:'youtube', youtubeId:yid, vimeoId:'', embedUrl:'', thumbnail:`https://i.ytimg.com/vi/${encodeURIComponent(yid)}/hqdefault.jpg`};
  const vid = vimeoId(src);
  if(vid) return {src, type:'vimeo', youtubeId:'', vimeoId:vid, embedUrl:'', thumbnail:''};
  if(isDirectVideoUrl(src)) return {src, type:'direct', youtubeId:'', vimeoId:'', embedUrl:'', thumbnail:''};
  if(isProbablyUrl(src)) return {src, type:'iframe', youtubeId:'', vimeoId:'', embedUrl:src, thumbnail: thumbnailFromEmbed(src)};
  return {src:'', type:'none', youtubeId:'', vimeoId:'', embedUrl:'', thumbnail:''};
}

function youtubeEmbedUrl(url, opts={}){
  const info = normalizeVideoUrl(url);
  if(info.type !== 'youtube' || !info.youtubeId) return '';
  let origin = 'https://www.youtube.com';
  try{ if(location.origin && /^https?:\/\//i.test(location.origin)) origin = location.origin; }catch(e){}
  const params = new URLSearchParams({
    playsinline:'1', rel:'0', modestbranding:'1',
    controls: String(opts.controls===false ? 0 : 1),
    autoplay: opts.autoplay ? '1' : '0',
    mute: opts.muted ? '1' : '0',
    iv_load_policy:'3', disablekb:'1',
    fs: String(opts.controls===false ? 0 : 1),
    cc_load_policy:'0', enablejsapi:'1',
    origin: origin, widget_referrer: origin
  });
  if(opts.loop){ params.set('loop','1'); params.set('playlist', info.youtubeId); }
  return 'https://www.youtube-nocookie.com/embed/' + encodeURIComponent(info.youtubeId) + '?' + params.toString();
}
function vimeoEmbedUrl(url, opts={}){
  const info = normalizeVideoUrl(url);
  if(info.type !== 'vimeo' || !info.vimeoId) return '';
  const params = new URLSearchParams({
    byline:'0', portrait:'0', title:'0', badge:'0', autopause:'0',
    autoplay: opts.autoplay ? '1' : '0',
    muted: opts.muted ? '1' : '0',
    controls: String(opts.controls===false ? 0 : 1),
    dnt:'1'
  });
  if(opts.loop) params.set('loop','1');
  return 'https://player.vimeo.com/video/' + encodeURIComponent(info.vimeoId) + '?' + params.toString();
}

function attachIframeEndedHandler(iframe, type, onEnded){
  let fired = false;
  const handler = event => {
    if(fired) return;
    if(event.source !== iframe.contentWindow) return;
    try{
      const raw = event.data;
      let data = raw;
      if(typeof raw === 'string'){ try { data = JSON.parse(raw); } catch(_) { return; } }
      if(!data || typeof data !== 'object') return;
      if(type === 'youtube' && data.event === 'onStateChange' && data.info === 0){
        fired = true; window.removeEventListener('message', handler); try{ onEnded(); }catch(_){}
      }
      if(type === 'vimeo' && (data.event === 'finish' || data.method === 'finish')){
        fired = true; window.removeEventListener('message', handler); try{ onEnded(); }catch(_){}
      }
    }catch(_){}
  };
  window.addEventListener('message', handler);
  iframe.__endedHandler = handler;
  return handler;
}

function thumbFor(v){
  if(!v) return '';
  const direct = safeUrl(v.thumbnail) || safeUrl(v.thumb) || safeUrl(v.image);
  if(direct) return direct;
  if(Array.isArray(v.links) && v.links.length){
    const first = v.links[0];
    let t = '';
    if(typeof first === 'string'){
      t = normalizeVideoUrl(first).thumbnail || '';
    } else if(first && typeof first === 'object'){
      const custom = safeUrl(first.thumbnail) || safeUrl(first.thumb) || safeUrl(first.image);
      if(custom) t = custom;
      else t = normalizeVideoUrl(first.url || first.link || first.src || '').thumbnail || '';
    }
    if(t) return t;
  }
  const info = normalizeVideoUrl(videoSource(v));
  return info.thumbnail || '';
}

async function performAuth(mode, email, password){
  email = String(email||'').trim().toLowerCase();
  password = String(password||'');
  if(!isValidEmail(email)) throw { code: 'auth/invalid-email' };
  if(password.length < 6) throw { code: 'auth/weak-password' };
  await initFirebaseOnce();
  let userCred;
  try{
    userCred = await firebaseEmailAuth(mode, email, password);
  }catch(err){
    if(mode === 'signin' && (err.code === 'auth/user-not-found' || err.code === 'auth/invalid-credential' || err.code === 'auth/invalid-login-credentials')){
      try{ userCred = await firebaseEmailAuth('signup', email, password); }catch(e2){ throw e2; }
    } else { throw err; }
  }

        email, uid, updatedAt: firebase.database.ServerValue.TIMESTAMP

    }catch(e){}
  }
  applyLocalProfile(profile);

    if(!app){ setTimeout(updateBgPreview, 100); return; }
    if(obj.type === 'image' && isValidBgUrl(obj.value)){
      const safe = obj.value.replace(/"/g,'&quot;');
      app.style.setProperty('background-image', `url("${safe}")`, '');
      app.style.setProperty('background-size','cover','');
      app.style.setProperty('background-position','center center','');
      app.style.setProperty('background-repeat','no-repeat','');
    } else if(obj.type === 'color'){
      app.style.backgroundColor = obj.value;
      document.documentElement.style.setProperty('--bg', obj.value);
    }
  }catch(e){}
  setTimeout(updateBgPreview, 100);
}

const HERO_AUTO_ADVANCE_MS = 3500;
let __heroAutoTimer = null;

function heroSlideThumb(slide){
  if(!slide) return '';
  const custom = safeUrl(slide.image) || safeUrl(slide.thumbnail) || safeUrl(slide.thumb) || safeUrl(slide.photo) || safeUrl(slide.backgroundImage);
  if(custom) return custom;
  const src = videoSource(slide);
  return normalizeVideoUrl(src).thumbnail || '';
}
function heroSlideLink(slide){
  if(!slide) return '';
  const raw = slide.openUrl || slide.openLink || slide.clickUrl || slide.link || slide.url || '';
  return safeUrl(raw);
}
function heroSlideLabel(slide){
  if(!slide) return '';
  return String(slide.t || slide.title || slide.name || '');
}
function heroSlideDesc(slide){
  if(!slide) return '';
  return String(slide.d || slide.description || slide.desc || '');
}

function heroSlideClick(i){
  const slide = HERO_SLIDES[i];
  if(!slide) return;
  const url = heroSlideLink(slide);
  if(!url){ showToast('Link not available', 'info'); return; }
  try{ window.open(url, '_blank', 'noopener,noreferrer'); }
  catch(e){ location.href = url; }
}

function renderHero(){
  const track = $('#heroTrack');
  const dots  = $('#heroDots');
  const heroEl = $('#homeHero');
  if(!track || !dots || !heroEl) return;

  if(__heroAutoTimer){ clearInterval(__heroAutoTimer); __heroAutoTimer = null; }

  const slides = HERO_SLIDES.filter(s => heroSlideThumb(s));
  if(!slides.length){
    heroEl.style.display = 'none';
    track.innerHTML = '';
    dots.innerHTML = '';
    return;
  }
  heroEl.style.display = '';

  track.innerHTML = slides.map((slide, i)=>{
    const thumb = heroSlideThumb(slide);
    const title = heroSlideLabel(slide);
    const desc  = heroSlideDesc(slide);
    const hasLink = !!heroSlideLink(slide);
    const hasText = !!(title || desc);
    return `<div class="hero-slide" data-hero-index="${i}">
      <div class="hero-card" onclick="heroSlideClick(${i})" role="button" tabindex="0">
        <div class="hero-media">
          <img src="${esc(thumb)}" alt="${esc(title||'Slide')}" loading="${i===0?'eager':'lazy'}"
               onerror="this.onerror=null;this.style.display='none';this.parentElement.classList.add('neutral');">
        </div>
        ${hasText ? `<div class="hero-scrim"></div>` : ''}
        ${hasText ? `<div class="hero-body">
          ${title ? `<div class="hero-title">${esc(title)}</div>` : ''}
          ${desc ? `<div class="hero-desc">${esc(desc)}</div>` : ''}
        </div>` : ''}
        ${hasLink ? `<div class="hero-link-hint">
          <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.4" stroke-linecap="round" stroke-linejoin="round"><path d="M14 4h6v6"/><path d="M20 4 10 14"/><path d="M20 14v5a1 1 0 0 1-1 1H5a1 1 0 0 1-1-1V5a1 1 0 0 1 1-1h5"/></svg>
          Open
        </div>` : ''}
      </div>
    </div>`;
  }).join('');

  dots.innerHTML = slides.map((_, i) =>
    `<button class="dot ${i===0?'on':''}" onclick="event.stopPropagation();heroGo(${i})" aria-label="Slide ${i+1}"></button>`
  ).join('');

  let idx = 0;
  const updateDots = () => Array.from(dots.children).forEach((d, i) => d.classList.toggle('on', i === idx));

  const goTo = (i, smooth) => {
    const n = Math.max(0, Math.min(slides.length - 1, Number(i) || 0));
    idx = n;
    updateDots();
    try{
      track.scrollTo({ left: track.clientWidth * n, behavior: smooth ? 'smooth' : 'auto' });
    }catch(e){
      track.scrollLeft = track.clientWidth * n;
    }
  };

  window.heroGo = i => { goTo(i, true); restartAuto(); };
  window.heroNext = () => {
    idx = (idx + 1) % slides.length;
    goTo(idx, true);
  };
  window.heroPrev = () => {
    idx = (idx - 1 + slides.length) % slides.length;
    goTo(idx, true);
  };
  window.HERO_SLIDES = slides;

  track.addEventListener('scroll', () => {
    const i = Math.round(track.scrollLeft / Math.max(1, track.clientWidth));
    if(i !== idx && i >= 0 && i < slides.length){
      idx = i;
      updateDots();
    }
  }, { passive: true });

  function restartAuto(){
    if(__heroAutoTimer){ clearInterval(__heroAutoTimer); __heroAutoTimer = null; }
    if(slides.length > 1 && HERO_AUTO_ADVANCE_MS > 0){
      __heroAutoTimer = setInterval(() => { window.heroNext(); }, HERO_AUTO_ADVANCE_MS);
    }
  }

  goTo(0, false);
  restartAuto();
}

let INDEX = [];
function searchAll(q){
  const s = q.toLowerCase().trim();
  if(!s) return [];
  const words = s.split(/\s+/);
  return INDEX.filter(item => {
    const hay = [item.title, item.desc, item.cat, ...(item.tags||[])].join(' ').toLowerCase();
    return words.every(w => hay.includes(w));
  });
}

function renderPlaylistDetail(){
  const p = currentPlaylist;
  if(!p) return;
  const links = Array.isArray(p.links) ? p.links : [];
  const count = p.n || links.length;
  const playlistUnlocked = !p.premium || isUnlocked(p.id);

  const titleEl = $('#pdTitle');
  if(titleEl) titleEl.textContent = p.t || 'Playlist';

  const metaEl = $('#pdInfoMeta');
  if(metaEl){
    const parts = [];
    if(count) parts.push(`${count} video${count>1?'s':''}`);
    if(p.cat) parts.push(esc(p.cat));
    if(p.dur) parts.push(esc(p.dur));
    if(p.premium && playlistUnlocked) parts.push('<span style="color:var(--green);font-weight:800">✓ Unlocked</span>');

  }
  const descEl = $('#pdDesc');
  if(descEl){
    if(p.desc){ descEl.textContent = p.desc; descEl.style.display = ''; }
    else { descEl.textContent = ''; descEl.style.display = 'none'; }
  }
  const listCountEl = $('#pdListCount');
  if(listCountEl) listCountEl.textContent = count ? `${count} video${count>1?'s':''}` : '';

  if(links.length){ renderPlaylistPlayer(links[0]); } else { renderPlaylistPlayer(null); }

  const listEl = $('#pdList');
  if(listEl){
    if(!links.length){
      listEl.innerHTML = `<div class="empty" style="padding:36px 24px">
        <div class="empty-ico">
          <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.7" stroke-linecap="round" stroke-linejoin="round"><path d="M3 6h13M3 12h13M3 18h9"/><path d="m17 14 5 3-5 3v-6Z"/></svg>
        </div>
        <h3>No videos yet</h3>
        <p>This playlist doesn't have any videos added.</p>
      </div>`;
    } else {
      listEl.innerHTML = links.map((link, i) => {
        const t = linkTitle(link, i);

        const creator = linkCreator(link) || p.creator || '';

          ? `<div class="pd-lock"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round" stroke-linejoin="round"><rect x="5" y="11" width="14" height="10" rx="2.5"/><path d="M8 11V7.5a4 4 0 0 1 8 0V11"/></svg></div>`

        const clickAttr = locked
          ? `unlockPremium('${esc(p.id)}','playlist','${esc(p.t)}',()=>playPlaylistVideo(${i}))`
          : `playPlaylistVideo(${i})`;
        return `<div class="pd-video-item ${active?'active':''}" onclick="${clickAttr}">
          <div class="pd-video-thumb">
            ${thumbContent}
            ${overlay}
            ${dur ? `<span class="pd-dur">${esc(dur)}</span>` : ''}
          </div>
          <div class="pd-video-info">
            <h4>${esc(t)}</h4>
            ${creator ? `<p class="pd-creator">${esc(creator)}</p>` : ''}

          </div>

  }
  const unlockWrap = $('#pdUnlockWrap');
  if(unlockWrap){
    if(p.premium && !playlistUnlocked){
      const price = (p.price != null && p.price !== '') ? Number(p.price) : null;
      unlockWrap.innerHTML = `

function renderPlaylistPlayer(link){
  const inner = $('#pdPlayerInner');
  if(!inner) return;
  const playlistUnlocked = !currentPlaylist || !currentPlaylist.premium || isUnlocked(currentPlaylist.id);
  const thisVideoLocked = !!(currentPlaylist && currentPlaylist.premium && !playlistUnlocked && currentPlaylistIndex > 0);
  if(thisVideoLocked){
    inner.innerHTML = `
      <div class="pd-empty" style="display:flex;flex-direction:column;gap:8px;text-align:center;padding:20px">
        <div style="font-size:34px">🔒</div>
        <div style="font-size:13px;opacity:.9">Premium content</div>
        <button class="btn btn-gold btn-sm" onclick="unlockPremium('${esc(currentPlaylist.id)}','playlist','${esc(currentPlaylist.t)}',()=>{ currentPlaylistIndex=${currentPlaylistIndex}; renderPlaylistPlayer((currentPlaylist.links||[])[${currentPlaylistIndex}]); renderPlaylistDetail(); })">Unlock to watch</button>
      </div>`;
    return;
  }
  const src=linkSrc(link);
  const title=linkTitle(link,currentPlaylistIndex);
  const info=normalizeVideoUrl(src);
  const thumb=thumbForLink(link);
  const playerSrc = info.src || src;
  const media = mountPlayerMedia(inner, playerSrc, {
    title, autoplay: !!playerSrc, muted: true, controls: true, eager: true, objectFit:'contain', loop: false,
    onEnded:()=>{
      const next=currentPlaylistIndex+1;
      if(currentPlaylist && next<(currentPlaylist.links||[]).length) playPlaylistVideo(next);
    },
    emptyText:'Video unavailable'
  });
  if(!media && thumb){
    inner.innerHTML=`<img src="${esc(thumb)}" alt="${esc(title)}" style="width:100%;height:100%;object-fit:cover;display:block">`;
  }
}
function playPlaylistVideo(index){
  if(!currentPlaylist) return;
  const links = currentPlaylist.links || [];
  if(index<0 || index>=links.length) return;
  const playlistUnlocked = !currentPlaylist.premium || isUnlocked(currentPlaylist.id);
  if(currentPlaylist.premium && !playlistUnlocked && index > 0){
    unlockPremium(currentPlaylist.id, 'playlist', currentPlaylist.t, () => {
      currentPlaylistIndex = index;
      renderPlaylistPlayer(links[index]);
      renderPlaylistDetail();
      $$('#pdList .pd-video-item').forEach((el,i)=>el.classList.toggle('active',i===index));
    });
    return;
  }
  currentPlaylistIndex = index;
  renderPlaylistPlayer(links[index]);
  $$('#pdList .pd-video-item').forEach((el,i)=>el.classList.toggle('active',i===index));
}

async function doLogin(){
  const emailEl = $('#loginEmail'), pwEl = $('#loginPassword'), errEl = $('#loginErr');
  const email = (emailEl?.value||'').trim();
  const password = (pwEl?.value||'');
  const btn = $('#loginSubmitBtn');
  const showErr = (msg, markEmail, markPw) => {
    if(errEl){ errEl.textContent = msg; errEl.style.display = 'flex'; }
    emailEl?.classList.toggle('error', !!markEmail);
    pwEl?.classList.toggle('error', !!markPw);
  };
  if(errEl){ errEl.style.display = 'none'; errEl.textContent=''; }
  emailEl?.classList.remove('error');
  pwEl?.classList.remove('error');
  if(!isValidEmail(email)){ showErr('Please enter a valid email address.', true, false); return; }
  if(password.length < 6){ showErr('Password must be at least 6 characters.', false, true); return; }
  if(btn){ btn.disabled = true; btn.dataset.orig = btn.textContent; btn.innerHTML = '<span class="spin"></span> Please wait…'; }
  try{
    await performAuth(__loginMode, email, password);
    showToast(__loginMode === 'signup' ? 'Account created!' : 'Signed in!', 'success');
    navStack = []; show('home'); lastTab = 'home';
    $$('.nav-item').forEach(n => n.classList.toggle('active', n.dataset.tab === 'home'));
    refreshPremiumUI();
  }catch(e){
    showErr(authErrorMessage(e.code) || e.message || 'Something went wrong.', e.code==='auth/invalid-email', e.code==='auth/weak-password' || e.code==='auth/wrong-password');
    if(btn){ btn.disabled = false; btn.textContent = btn.dataset.orig || 'Try again'; }
  }
}

function firstValue(o, keys, fallback=''){
  for(const k of keys){ if(o && o[k] !== undefined && o[k] !== null && o[k] !== '') return o[k]; }
  return fallback;
}
function normalizeUrl(v){ return typeof v === 'string' ? v.trim() : ''; }
function videoSource(v){ return normalizeVideoUrl(v).src; }
function normalizeCollection(data){
  if(!data) return [];
  if(Array.isArray(data)) return data.map((v,i)=>({id:String(i), value:v}));
  return Object.keys(data).map(k => ({id:k, value:data[k]}));
}
function playlistLinks(p){
  let links = firstValue(p||{}, ['videos','videoLinks','links','items','lessons'], []);
  if(!links){
    const single = firstValue(p||{}, ['videoUrl','videoURL','videoLink','video','url','link','playlistUrl'], '');
    links = single ? [single] : [];
  }
  if(typeof links === 'string') links = links.split(/\r?\n|,|;\s*(?=https?:\/\/)/).map(x=>x.trim()).filter(Boolean);
  if(links && typeof links === 'object' && !Array.isArray(links)) links = Object.values(links);
  if(!Array.isArray(links)) return [];
  return links.map(x => {
    if(typeof x === 'string') return x;
    return {
      url: videoSource(x) || normalizeUrl(firstValue(x, ['url','link','src'], '')),
      title: String(firstValue(x, ['title','t','name','videoName'], '')),
      dur: String(firstValue(x, ['dur','duration'], '')),
      creator: String(firstValue(x, ['creator','channel','author','uploader'], '')),
      thumbnail: normalizeUrl(firstValue(x, ['thumbnail','thumb','image','imageUrl','photo','photoUrl'], ''))
    };
  }).filter(l => {
    if(typeof l === 'string') return !!l;
    return !!l.url;
  });
}
function makeVideo(raw, id, index){
  const v = raw || {};
  const src = videoSource(v);
  const vc = Number(firstValue(v, ['views','viewCount'], 0)) || 0;
  const title = String(firstValue(v, ['t','title','name','videoName'], ''));
  if(!title && !src) return null;
  let viewsText = '';
  if(vc){
    viewsText = vc >= 1000000 ? (vc/1000000).toFixed(1).replace('.0','')+'M'
      : vc >= 1000 ? (vc/1000).toFixed(vc>=100000?0:1).replace('.0','')+'K' : String(vc);
  }
  return {
    id: String(firstValue(v, ['id'], id || ('v'+(index+1)))),
    t: title || 'Untitled Video',
    cat: String(firstValue(v, ['cat','category'], '')),
    dur: String(firstValue(v, ['dur','duration'], '')),
    views: viewsText,
    ago: String(firstValue(v, ['ago','time','createdText'], '')),
    premium: !!firstValue(v, ['premium','isPremium','paid'], false),
    tags: Array.isArray(v.tags) ? v.tags : [],
    desc: String(firstValue(v, ['desc','description','videoDescription'], '')),
    link: src,
    thumbnail: normalizeUrl(firstValue(v, ['thumbnail','thumbnailUrl','image','imageUrl','photo','photoUrl','thumb'], '')),
    createdAt: Number(firstValue(v, ['createdAt','created','timestamp'], 0)) || 0
  };
}
function makePlaylist(raw, id, index){
  const p = raw || {};
  const links = playlistLinks(p);
  const count = Number(firstValue(p, ['n','count','videoCount'], links.length)) || links.length;
  const title = String(firstValue(p, ['t','title','name','playlistName'], ''));
  if(!title && !links.length) return null;
  return {
    id: String(firstValue(p, ['id'], id || ('p'+(index+1)))),
    t: title || 'Untitled Playlist',
    n: count,
    dur: String(firstValue(p, ['dur','duration','totalDuration'], '')),
    premium: !!firstValue(p, ['premium','isPremium','paid'], false),
    cat: String(firstValue(p, ['cat','category'], '')),
    progress: Number(firstValue(p, ['progress'], 0)) || 0,
    desc: String(firstValue(p, ['desc','description','playlistDescription'], '')),
    thumbnail: normalizeUrl(firstValue(p, ['thumbnail','thumbnailUrl','image','imageUrl','photo','photoUrl','thumb'], '')),
    price: firstValue(p, ['price'], null),
    old: firstValue(p, ['old','oldPrice'], null),
    creator: String(firstValue(p, ['creator','author','channel','uploader'], '')),
    createdAt: Number(firstValue(p, ['createdAt','created','timestamp'], 0)) || 0,
    links: links
  };
}
function makeApp(raw, id, index){
  const a = raw || {};
  const name = String(firstValue(a, ['n','name','title','appName'], ''));
  if(!name) return null;
  const screenshots = Array.isArray(a.screenshots) ? a.screenshots.filter(Boolean).map(x=>normalizeUrl(x)).filter(Boolean) : [];
  return {
    id: String(firstValue(a, ['id'], id || ('a'+(index+1)))),
    n: name,
    cat: String(firstValue(a, ['cat','category'], '')),
    rating: firstValue(a, ['rating'], ''),
    ver: String(firstValue(a, ['ver','version'], '')),
    size: String(firstValue(a, ['size'], '')),
    dl: String(firstValue(a, ['dl','downloads'], '')),
    tag: String(firstValue(a, ['tag','tagline','shortDescription'], '')),
    desc: String(firstValue(a, ['desc','description','longDesc'], '')),
    shortDesc: String(firstValue(a, ['shortDesc','shortDescription'], '')),
    features: Array.isArray(a.features) ? a.features : [],
    whatsnew: String(firstValue(a, ['whatsnew','whatsNew','releaseNotes'], '')),
    developer: String(firstValue(a, ['developer','publisher'], '')),
    image: normalizeUrl(firstValue(a, ['image','icon','logo','imageUrl','logoUrl'], '')),
    logoUrl: normalizeUrl(firstValue(a, ['logoUrl','logo','icon','image','imageUrl'], '')),
    banner: normalizeUrl(firstValue(a, ['banner','bannerUrl'], '')),
    installUrl: normalizeUrl(firstValue(a, ['installUrl','installURL','downloadUrl','openUrl','storeUrl'], '')),
    screenshots,
    featured: !!firstValue(a, ['featured','isFeatured'], false),
    createdAt: Number(firstValue(a, ['createdAt','created','timestamp'], 0)) || 0
  };
}
function makeSlider(raw, i){
  const x = raw || {};
  const sourceRaw = typeof x === 'string' ? x : firstValue(x, ['videoUrl','videoURL','videoLink','video','src','url','link','embed','embedUrl','iframeUrl'], '');
  const image = typeof x === 'string' ? '' : normalizeUrl(firstValue(x, ['image','imageUrl','photo','photoUrl','backgroundImage','thumbnail','thumbnailUrl','thumb'], ''));
  const title = typeof x === 'string' ? 'Learnify' : String(firstValue(x, ['t','title','name'], ''));
  const openUrl = normalizeUrl(firstValue(x, ['openUrl','openLink','clickUrl'], ''));
  const videoUrl = normalizeVideoUrl(sourceRaw || x).src;
  const videoId = String(firstValue(x, ['videoId','targetVideo'], ''));
  const desc = String(firstValue(x, ['d','description','desc'], ''));
  const badge = String(firstValue(x, ['badge','label','type'], ''));
  const cta = String(firstValue(x, ['cta','buttonText'], ''));
  if(!title && !image && !videoUrl && !videoId && !openUrl) return null;
  return {
    id: String(firstValue(x, ['id'], 's'+(i+1))),
    g: String(firstValue(x, ['g','gradient'], 'g'+((i%10)+1))),
    badge,
    premium: !!firstValue(x, ['premium','isPremium','paid'], false),
    t: title || 'Learnify',
    d: desc,
    cta,
    image, videoUrl, openUrl, videoId
  };
}

function applyAppSettings(settings){
  if(!settings) return;
  try{
    const bg = firstValue(settings, ['backgroundColor','background','bg','appBackground'], '');
    const img = firstValue(settings, ['backgroundImage','backgroundPhoto','backgroundUrl','bgImage','imageUrl','photoUrl'], '');
    const mode = String(firstValue(settings, ['backgroundMode','bgMode'], img ? 'image' : 'color') || '').toLowerCase();
    const app = $('#app');
    const validColor = /^(#[0-9a-f]{3,8}|rgb\(|rgba\(|hsl\(|hsla\()/i.test(String(bg));
    if(validColor){
      document.documentElement.style.setProperty('--bg', String(bg));
      if(app) app.style.backgroundColor = String(bg);
      document.body.style.backgroundColor = String(bg);
    }
    if(app){
      if(img && mode !== 'color'){
        const safeImg = String(img).replace(/\"/g,'&quot;');
        app.style.setProperty('background-image', `url("${safeImg}")`, '');
        app.style.setProperty('background-size','cover','');
        app.style.setProperty('background-position','center center','');
        app.style.setProperty('background-repeat','no-repeat','');
        app.style.setProperty('background-attachment','scroll','');
      }else if(!img && mode === 'color'){
        app.style.removeProperty('background-image');
        app.style.backgroundImage = 'none';
      }
    }
    setTimeout(updateBgPreview, 100);
  }catch(e){ console.error('Background apply failed:',e); }
  const meImage = firstValue(settings, ['meIconImage','meIconImageUrl','meIcon','profileIconImage','userIconImage'], '');
  const meLink = firstValue(settings, ['meIconClickLink','meIconLink','profileIconLink','userIconLink'], '');
  if(meImage){
    const av = $('.avatar-btn');
    if(av){
      av.innerHTML = `<img src="${esc(meImage)}" alt="Me" style="width:100%;height:100%;object-fit:cover;border-radius:14px">`;
      av.onclick = () => meLink ? window.open(meLink,'_blank','noopener,noreferrer') : go('profile');
    }
  }
}
